const LLAVE = "";          // la llave anon (public), entre las comillas
const CODIGO = "";         // tu código secreto, entre las comillas

const URL_APP = "https://vmmterminus-star.github.io/agenda-uni/";

/* ---------- letra: Outfit (instalada con iFont). Si el iPhone no la encuentra,
   usa la redondeada del sistema. Para regresar a la redondeada, cambia true por false. ---------- */
const USAR_OUTFIT = true;
const OUTFIT = { regular:"Outfit-Regular", medium:"Outfit-Medium", semibold:"Outfit-SemiBold", bold:"Outfit-Bold" };
function fuente(peso, tam){
  if(USAR_OUTFIT){ try{ const f = new Font(OUTFIT[peso], tam); if(f) return f; }catch(e){} }
  return Font[peso + "RoundedSystemFont"](tam);
}

/* ---------- colores de la agenda (claro / oscuro) ---------- */
function dyn(a, b){ return Color.dynamic(new Color(a), new Color(b)); }
const C = {
  bg:    dyn("#FFFFFF", "#23201D"),
  txt:   dyn("#33302C", "#EDE7DF"),
  mut:   dyn("#7A7267", "#A79E92"),
  hint:  dyn("#A79E92", "#7A7267"),
  acc:   dyn("#C97B5A", "#D9906F"),
  soft:  dyn("#FBF9F5", "#2E2A26"),
  line:  dyn("#E9E3DA", "#3A3530"),
  danbg: dyn("#FBE9E4", "#4A2A24"), dantx: dyn("#8C3325", "#F0A594"), dan: dyn("#B5503F", "#E08472"),
  ambbg: dyn("#F0E8D8", "#453A26"), ambtx: dyn("#6E5522", "#E3C98F"), amb: dyn("#A98545", "#D4B06C"),
  grn:   dyn("#5F8A7D", "#8DBBAE")
};
const F = {
  tit:  fuente("bold", 11),
  big:  fuente("bold", 34),
  fila: fuente("regular", 12),
  fec:  fuente("medium", 11),
  cls:  fuente("semibold", 13),
  sm:   fuente("regular", 11),
  band: fuente("bold", 9)
};

/* ---------- fechas: se calculan aquí, al momento ---------- */
const DIAS   = ["dom","lun","mar","mié","jue","vie","sáb"];
const DIAS_L = ["Domingo","Lunes","Martes","Miércoles","Jueves","Viernes","Sábado"];
const MESC   = ["ene","feb","mar","abr","may","jun","jul","ago","sep","oct","nov","dic"];
const MS = 86400000;
function pd(s){ const p = s.split("-"); return new Date(+p[0], +p[1]-1, +p[2]); }
function hoy0(){ const d = new Date(); d.setHours(0,0,0,0); return d; }
function fmt(d){ return d.getFullYear()+"-"+("0"+(d.getMonth()+1)).slice(-2)+"-"+("0"+d.getDate()).slice(-2); }
function dif(s){ return Math.round((pd(s) - hoy0()) / MS); }
function ahoraHM(){ const d = new Date(); return ("0"+d.getHours()).slice(-2)+":"+("0"+d.getMinutes()).slice(-2); }

function etiqueta(T){
  const d = dif(T.f), h = T.h ? " "+T.h : "";
  if(d < 0)  return d === -1 ? "ayer" : (-d)+" días tarde";
  if(d === 0) return "hoy"+h;
  if(d === 1) return "mañana"+h;
  const x = pd(T.f);
  return d <= 6 ? DIAS[x.getDay()]+" "+x.getDate() : x.getDate()+" "+MESC[x.getMonth()];
}
function nivel(T){ const d = dif(T.f); return d < 0 ? "atr" : (d <= 1 ? "hoy" : "no"); }

/* la siguiente clase que todavía no termina */
function siguientesClases(R){
  const hs = fmt(hoy0()), hm = ahoraHM();
  const cla = (R.cla || []).filter(c => c.d > hs || (c.d === hs && (c.hf || c.hi || "99") > hm));
  return cla;
}
function cuandoClase(c){
  const d = dif(c.d);
  if(d === 0) return c.hi;
  if(d === 1) return "mañana "+c.hi;
  return DIAS[pd(c.d).getDay()]+" "+c.hi;
}

/* estado de Proyecto 9, igual que la ruta de la app */
function rutaP9(R){
  const s1 = pd(R.s1), h = hoy0();
  const lunes = new Date(h); lunes.setDate(h.getDate() - ((h.getDay()+6) % 7));
  const sem = Math.max(1, Math.min(17, Math.round((lunes - s1) / (7*MS)) + 1));
  let n = 1, nombre = "";
  (R.et || []).forEach((E, i) => { if(E.a <= sem && sem <= E.b){ n = i+1; nombre = E.n; } });
  let atraso = 0;
  (R.tar || []).forEach(T => { if(!T.e) return; const d = dif(T.f); if(d < 0) atraso = Math.max(atraso, Math.floor(-d/7)); });
  if(atraso >= 2) return { n, nombre, total:(R.et||[]).length, txt:"Atraso de "+atraso+" sem", col:C.dan };
  if(atraso === 1) return { n, nombre, total:(R.et||[]).length, txt:"En riesgo", col:C.amb };
  return { n, nombre, total:(R.et||[]).length, txt:"Al día", col:C.grn };
}

/* ---------- piezas ---------- */
function titulo(p, txt){
  const t = p.addText(txt.toUpperCase()); t.font = F.tit; t.textColor = C.acc; t.lineLimit = 1;
  return t;
}
function texto(p, txt, font, color, lineas){
  const t = p.addText(txt); t.font = font; t.textColor = color; t.lineLimit = lineas || 1;
  return t;
}
function fila(p, T){
  const nv = nivel(T);
  const r = p.addStack(); r.layoutHorizontally(); r.centerAlignContent();
  r.setPadding(3, 5, 3, 6); r.cornerRadius = 6;
  r.backgroundColor = nv === "atr" ? C.danbg : (nv === "hoy" ? C.ambbg : C.soft);
  const s = r.addStack(); s.size = new Size(3, 13); s.cornerRadius = 1.5; s.backgroundColor = new Color(T.c || "#A79E92");
  r.addSpacer(5);
  texto(r, T.t, F.fila, C.txt);
  r.addSpacer();
  r.addSpacer(4);
  texto(r, etiqueta(T), F.fec, nv === "atr" ? C.dantx : (nv === "hoy" ? C.ambtx : C.mut));
  p.addSpacer(3);
}
/* encabezado de semana: letra chica y una línea del color de los encabezados de tu agenda */
const ROJO_SEM = "#8A4529", ROJO_URGE = "#8C3325";
function cabecera(p, txt, urge){
  const hex = urge ? ROJO_URGE : ROJO_SEM;
  p.addSpacer(4);
  texto(p, txt, fuente("semibold", 9), dyn(hex, "#E7A68C"));
  p.addSpacer(1);
  const dc = new DrawContext();
  dc.size = new Size(600, 2); dc.opaque = false; dc.respectScreenScale = true;
  dc.setFillColor(new Color(hex)); dc.fillRect(new Rect(0, 0, 600, 2));
  const img = p.addImage(dc.getImage()); img.resizable = true;
  p.addSpacer(3);
}

/* ---------- los tres tamaños ---------- */
function chico(w, R){
  const hs = fmt(hoy0());
  const hoyN = R.tar.filter(T => T.f === hs).length;
  const atr  = R.tar.filter(T => T.f < hs).length;
  titulo(w, "Hoy");
  w.addSpacer(2);
  texto(w, String(hoyN), fuente("bold", 30), C.txt);
  let sub = hoyN === 1 ? "pendiente" : "pendientes";
  if(!hoyN) sub = "nada vence hoy";
  if(atr) sub += " · " + atr + (atr === 1 ? " atrasada" : " atrasadas");
  texto(w, sub, F.sm, atr ? C.dantx : C.mut);
  w.addSpacer();
  const sig = siguientesClases(R)[0];
  texto(w, "Siguiente clase", F.sm, C.mut);
  if(sig){
    texto(w, sig.n, F.cls, C.txt);
    texto(w, cuandoClase(sig) + (sig.a ? " · " + sig.a : ""), F.sm, C.mut);
  } else texto(w, "ninguna pronto", F.cls, C.txt);
}

function mediano(w, R){
  const cab = w.addStack(); cab.layoutHorizontally();
  titulo(cab, "Tareas"); cab.addSpacer();
  const lista = R.tar.slice(0, 4);
  if(R.tar.length > 4) texto(cab, "+" + (R.tar.length - 4) + " más", F.sm, C.hint);
  w.addSpacer(6);
  if(!lista.length) texto(w, "Nada pendiente. Vas al corriente.", F.fila, C.mut);
  lista.forEach(T => fila(w, T));
  w.addSpacer();
}

function grande(w, R){
  const h = hoy0();
  titulo(w, DIAS_L[h.getDay()] + " " + h.getDate());
  w.addSpacer(4);
  const cls = siguientesClases(R), hs = fmt(h);
  const hoyC = cls.filter(c => c.d === hs), luego = cls.find(c => c.d > hs);
  if(hoyC.length){
    const c = hoyC[0];
    texto(w, c.n + " " + c.hi + (c.hf ? "–" + c.hf : ""), F.cls, C.txt);
    const extra = [];
    if(c.a) extra.push(c.a);
    if(hoyC[1]) extra.push("luego " + hoyC[1].n + " " + hoyC[1].hi);
    else if(luego) extra.push(cuandoClase(luego) + " " + luego.n);
    texto(w, extra.join(" · "), F.sm, C.mut);
  } else {
    texto(w, "Sin más clases hoy", F.cls, C.txt);
    if(luego) texto(w, "Siguiente: " + luego.n + " " + cuandoClase(luego), F.sm, C.mut);
  }
  w.addSpacer(4);

  /* mismos grupos que el apartado Tareas de tu Inicio */
  const dom = new Date(h); dom.setDate(h.getDate() + (7 - h.getDay()) % 7);
  const domS = fmt(dom), s1 = pd(R.s1);
  const lunesDe = d => { const x = new Date(d); x.setDate(x.getDate() - ((x.getDay()+6) % 7)); return x; };
  const semDe = f => { const n = Math.round((lunesDe(pd(f)) - s1) / (7*MS)) + 1; return (n >= 1 && n <= 17) ? n : null; };
  const semHoy = Math.max(1, Math.min(17, Math.round((lunesDe(h) - s1) / (7*MS)) + 1));
  const rango = n => {
    const a = new Date(s1); a.setDate(a.getDate() + (n-1)*7);
    const b = new Date(a); b.setDate(b.getDate() + 6);
    return a.getDate() + (a.getMonth() !== b.getMonth() ? " " + MESC[a.getMonth()] : "") + "–" + b.getDate() + " " + MESC[b.getMonth()];
  };
  const grupoDe = T => {
    if(T.f < hs)  return { k:"00", t:"Atrasado", urge:true };
    if(T.f === hs) return { k:"01", t:"Hoy", urge:true };
    if(T.f <= domS) return { k:"02", t:"Resto de esta semana" };
    const sn = semDe(T.f);
    if(!sn) return { k:"yy", t:"Después del semestre" };
    return { k:"s" + ("0"+sn).slice(-2), t:(sn === semHoy+1 ? "Próxima semana · S" + sn : "Semana " + sn) + " · " + rango(sn) };
  };
  const grupos = [];
  R.tar.forEach(T => {
    const G = grupoDe(T);
    let g = grupos.find(x => x.k === G.k);
    if(!g){ g = Object.assign({ ts:[] }, G); grupos.push(g); }
    g.ts.push(T);
  });
  grupos.sort((a, b) => a.k < b.k ? -1 : 1);

  /* caben ~7 renglones; cada encabezado ocupa casi lo mismo que media tarea */
  let usado = 0;
  const TOPE = 8.5;
  if(!R.tar.length){ w.addSpacer(8); texto(w, "Nada pendiente. Vas al corriente.", F.fila, C.mut); }
  for(const g of grupos){
    if(usado + 0.6 + 1 > TOPE) break;
    cabecera(w, g.t, g.urge);
    usado += 0.6;
    for(const T of g.ts){
      if(usado + 1 > TOPE) break;
      fila(w, T); usado += 1;
    }
  }
  w.addSpacer();

  const P = rutaP9(R);
  const pie = w.addStack(); pie.layoutHorizontally();
  texto(pie, "Proyecto 9 · etapa " + P.n + " de " + P.total, F.sm, C.mut);
  pie.addSpacer();
  texto(pie, P.txt, fuente("bold", 11), P.col);
}

/* pantalla de bloqueo: por si algún día lo pones ahí */
function bloqueo(w, R, fam){
  const hs = fmt(hoy0());
  const hoyN = R.tar.filter(T => T.f === hs).length, atr = R.tar.filter(T => T.f < hs).length;
  const sig = siguientesClases(R)[0];
  if(fam === "accessoryCircular"){
    texto(w, String(hoyN), fuente("bold", 22), Color.white()).centerAlignText();
    texto(w, "hoy", fuente("regular", 10), Color.white()).centerAlignText();
    return;
  }
  const res = hoyN + " para hoy" + (atr ? " · " + atr + " atrasada" + (atr > 1 ? "s" : "") : "");
  if(fam === "accessoryInline"){ texto(w, res + (sig ? " · " + sig.n + " " + cuandoClase(sig) : ""), fuente("regular", 12), Color.white()); return; }
  texto(w, res, fuente("bold", 13), Color.white());
  if(sig) texto(w, sig.n + " · " + cuandoClase(sig), fuente("regular", 12), Color.white());
  const pr = R.tar.find(T => T.f >= hs);
  if(pr) texto(w, pr.t, fuente("regular", 12), Color.white());
}

/* ---------- datos: de la nube, y si no hay internet, lo último guardado ---------- */
function limpiaUrl(u){
  u = (u || "").trim().replace(/\s+/g, "").replace(/\/+$/, "").replace(/\/rest(\/v1)?$/i, "").replace(/\/+$/, "");
  if(u && !/^https?:\/\//i.test(u)) u = "https://" + u;
  return u;
}
async function cargar(){
  const fm = FileManager.local();
  const ruta = fm.joinPath(fm.documentsDirectory(), "agenda-uni-widget.json");
  /* lo guardado lleva los datos y las dos horas; la versión vieja solo traía los datos */
  const guardado = () => {
    if(!fm.fileExists(ruta)) return null;
    const g = JSON.parse(fm.readString(ruta));
    return (g && g.data && g.leido) ? g : { data: g, at: null, leido: null };
  };
  const base = limpiaUrl(SUPABASE_URL);
  if(!/^https:\/\/[a-z0-9-]+\.supabase\.co$/i.test(base))
    return { error: "La URL de arriba no parece de Supabase. Debe verse así: https://algo.supabase.co  ·  La tuya: " + (SUPABASE_URL || "(vacía)") };
  let txt;
  try{
    const r = new Request(base + "/rest/v1/escuela_sync?code=eq." + encodeURIComponent(CODIGO.trim() + "__widget") + "&select=data,updated_at");
    r.headers = { apikey: LLAVE.trim(), Authorization: "Bearer " + LLAVE.trim() };
    r.timeoutInterval = 12;
    txt = await r.loadString();
  } catch(e){
    const g = guardado();
    if(g) return { R: g.data, at: g.at, leido: g.leido, sinRed: true };
    return { error: "No pude conectarme a Supabase y todavía no hay nada guardado. Detalle: " + (e && e.message ? e.message : e) };
  }
  let rows;
  try{ rows = JSON.parse(txt); }
  catch(e){ return { error: "Supabase respondió algo raro. Revisa la URL de arriba. Respuesta: " + String(txt).slice(0, 80) }; }
  if(Array.isArray(rows) && rows[0] && rows[0].data){
    /* lo que ya palomeaste en tu agenda personal no sale, aunque aún no abras la escolar */
    try{
      const rh = new Request(base + "/rest/v1/escuela_sync?code=like." + encodeURIComponent(CODIGO.trim() + "__hecha__") + "*&select=data");
      rh.headers = { apikey: LLAVE.trim(), Authorization: "Bearer " + LLAVE.trim() };
      rh.timeoutInterval = 8;
      const hs = await rh.loadJSON();
      if(Array.isArray(hs)){
        const fuera = hs.filter(x => x.data && x.data.st === 2).map(x => x.data.i);
        if(fuera.length) rows[0].data.tar = (rows[0].data.tar || []).filter(T => fuera.indexOf(T.i) < 0);
      }
    } catch(e){}
    const g = { data: rows[0].data, at: rows[0].updated_at || null, leido: Date.now() };
    fm.writeString(ruta, JSON.stringify(g));
    return { R: g.data, at: g.at, leido: g.leido };
  }
  if(Array.isArray(rows)) return { error: "Todavía no hay resumen con ese código. Revisa que el código sea igual al de tu agenda, abre la agenda y espera la pastilla verde." };
  return { error: "Supabase no aceptó la llave. Revisa la llave de arriba. Detalle: " + String(rows && (rows.message || rows.msg || rows.error) || "").slice(0, 80) };
}

/* ---------- la línea de abajo: cuándo leyó el widget y cuándo mandó cambios tu agenda ---------- */
function haceCuanto(t){
  const min = Math.floor((Date.now() - new Date(t).getTime()) / 60000);
  if(min < 1) return "hace un momento";
  if(min < 60) return "hace " + min + " min";
  const h = Math.floor(min / 60);
  if(h < 24) return "hace " + h + " h";
  const d = Math.floor(h / 24);
  return d === 1 ? "hace 1 día" : "hace " + d + " días";
}
function hhmm(t){ const d = new Date(t); return ("0"+d.getHours()).slice(-2) + ":" + ("0"+d.getMinutes()).slice(-2); }
function lineaInfo(w, res, corto){
  let s = res.sinRed ? "Sin señal · " : "";
  s += res.leido ? "Leído " + hhmm(res.leido) : "Leído antes";
  if(!corto && res.at) s += " · tu agenda mandó cambios " + haceCuanto(res.at);
  w.addSpacer(3);
  texto(w, s, fuente("regular", 9), res.sinRed ? C.dantx : C.hint);
}

/* ---------- armar ---------- */
const fam = config.widgetFamily || "large";
const w = new ListWidget();
w.url = URL_APP;
w.refreshAfterDate = new Date(Date.now() + 15*60*1000);
const esBloqueo = fam.indexOf("accessory") === 0;
if(!esBloqueo){ w.backgroundColor = C.bg; w.setPadding(14, 14, 14, 14); }

if(!SUPABASE_URL || !LLAVE || !CODIGO){
  texto(w, "Falta poner tus datos arriba del script: la URL, la llave y tu código.", esBloqueo ? fuente("regular", 12) : F.fila, esBloqueo ? Color.white() : C.txt, 7);
} else {
  const res = await cargar();
  if(res.error) texto(w, res.error, esBloqueo ? fuente("regular", 12) : F.fila, esBloqueo ? Color.white() : C.txt, 7);
  else {
    const R = res.R; R.tar = R.tar || [];
    if(esBloqueo) bloqueo(w, R, fam);
    else if(fam === "small"){ chico(w, R); lineaInfo(w, res, true); }
    else if(fam === "medium"){ mediano(w, R); lineaInfo(w, res, false); }
    else { grande(w, R); lineaInfo(w, res, false); }
  }
}

if(config.runsInWidget) Script.setWidget(w);
else await w.presentLarge();
Script.complete();
